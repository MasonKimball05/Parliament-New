# Parliament crontab
# Edit with: sudo crontab -e
# View with: sudo crontab -l
#
# Logs go to /var/log/parliament/
# Make sure the log dir exists: sudo mkdir -p /var/log/parliament
#
# Set this so the watchdog jobs below can email you (any non-empty output from
# a cron job is mailed here — independent of the app's own SMTP):
# MAILTO=mason.kimball05@gmail.com

# ---------------------------------------------------------------------------
# Uptime / 502 watchdog — external, every 5 min (v3.15.2)
#
# Emails once if the site has been failing for ~15 min (3 consecutive checks)
# and once when it recovers. Plain shell + system cron on purpose: it must
# survive a dead Daphne (the 07-19 502). Hits the origin through nginx so a
# Cloudflare-cached 200 can't mask an origin outage.
# ---------------------------------------------------------------------------
*/5 * * * * bash /var/www/Parliament-New/scripts/check_uptime.sh

# ---------------------------------------------------------------------------
# Scheduled announcements — every 15 minutes
# ---------------------------------------------------------------------------
*/15 * * * * cd /var/www/Parliament-New && echo "=== $(date) ===" >> /var/log/parliament/scheduled_announcements.log && source venv/bin/activate && python manage.py process_scheduled_announcements >> /var/log/parliament/scheduled_announcements.log 2>&1

# ---------------------------------------------------------------------------
# Session cleanup — weekly, Sunday 3am
# ---------------------------------------------------------------------------
0 3 * * 0 cd /var/www/Parliament-New && source venv/bin/activate && python manage.py cleanup_sessions >> /var/log/parliament/cleanup.log 2>&1

# ---------------------------------------------------------------------------
# Database backups
#
# Backups are saved to /var/backups/parliament/ (12 kept, older ones pruned).
# Set PARLIAMENT_BACKUP_DIR in /etc/environment if you want a different path.
# To restore: sudo ./scripts/restore_db.sh /var/backups/parliament/<filename>.dump
# ---------------------------------------------------------------------------

# Weekly backup — Sunday 2am, September through May
0 2 * 1-5,9-12 0 cd /var/www/Parliament-New && source venv/bin/activate && python manage.py backup_db >> /var/log/parliament/backup.log 2>&1

# Monthly backup — 1st of month, 2am, June through August
0 2 1 6-8 * cd /var/www/Parliament-New && source venv/bin/activate && python manage.py backup_db >> /var/log/parliament/backup.log 2>&1

# ---------------------------------------------------------------------------
# Daily digest watchdog — independent of Celery (v3.15.2)
#
# Alerts if the daily site digest hasn't successfully sent in ~26h — the fix
# for the 07-15 incident where the digest silently stopped for 4 days. Runs
# from SYSTEM cron on purpose: it must survive a dead Celery worker/beat.
# Set MAILTO at the top of the crontab so a non-zero exit emails you via a
# channel that doesn't depend on Django's SMTP:
#   MAILTO=mason.kimball05@gmail.com
# Run at noon UTC, after the 09:30 UTC digest window.
# ---------------------------------------------------------------------------
0 12 * * * cd /var/www/Parliament-New && source venv/bin/activate && python manage.py check_digest_freshness
